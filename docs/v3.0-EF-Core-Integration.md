# GameStore API v3.0: Entity Framework Core Integration

## Overview

This document outlines the implementation plan for v3.0 of the GameStore API, which integrates Entity Framework Core (EF Core) to replace in-memory collections with a SQL Server database for persistent storage.

## Key Benefits of EF Core Integration

1. **Data Persistence**: Data survives application restarts
2. **Complex Queries**: Support for advanced querying capabilities
3. **Transactions**: ACID compliance for data operations
4. **Migrations**: Structured schema evolution
5. **Performance**: Optimized data access with caching and lazy loading

### 2. Create DbContext

```csharp
using GameStore.Api.Models;
using Microsoft.EntityFrameworkCore;

namespace GameStore.Api.Data;

public class GameStoreContext : DbContext
{
    public GameStoreContext(DbContextOptions<GameStoreContext> options)
        : base(options)
    {
    }

    public DbSet<Game> Games => Set<Game>();
    public DbSet<Genre> Genres => Set<Genre>();

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        modelBuilder.Entity<Game>(entity =>
        {
            entity.HasKey(e => e.Id);
            entity.Property(e => e.Name).IsRequired().HasMaxLength(50);
            entity.Property(e => e.Price).HasColumnType("decimal(18,2)");
            entity.Property(e => e.ImageUrl).HasMaxLength(200);
            entity.Property(e => e.Description).HasMaxLength(2000);

            // Configure relationship
            entity.HasOne(e => e.Genre)
                  .WithMany(g => g.Games)
                  .HasForeignKey(e => e.GenreId)
                  .OnDelete(DeleteBehavior.Restrict);
        });

        modelBuilder.Entity<Genre>(entity =>
        {
            entity.HasKey(e => e.Id);
            entity.Property(e => e.Name).IsRequired().HasMaxLength(50);
        });

        // Seed data
        SeedData.ConfigureModelBuilder(modelBuilder);
    }
}
```

### 4. Implement EF Core Repositories

```csharp
public class EfGameRepository : IGameRepository
{
    private readonly GameStoreContext _context;

    public EfGameRepository(GameStoreContext context)
    {
        _context = context;
    }

    public async Task<IEnumerable<Game>> GetAllAsync()
    {
        return await _context.Games.AsNoTracking().ToListAsync();
    }

    public async Task<Game?> GetByIdAsync(Guid id)
    {
        return await _context.Games
            .AsNoTracking()
            .FirstOrDefaultAsync(g => g.Id == id);
    }

    public async Task<Game> CreateAsync(Game game)
    {
        _context.Games.Add(game);
        await _context.SaveChangesAsync();
        return game;
    }

    public async Task UpdateAsync(Game game)
    {
        _context.Entry(game).State = EntityState.Modified;
        try
        {
            await _context.SaveChangesAsync();
        }
        catch (DbUpdateConcurrencyException)
        {
            if (!await GameExistsAsync(game.Id))
            {
                throw new KeyNotFoundException($"Game with ID {game.Id} not found");
            }
            throw;
        }
    }

    public async Task DeleteAsync(Guid id)
    {
        var game = await _context.Games.FindAsync(id);

        if (game is null)
        {
            throw new KeyNotFoundException($"Game with ID {id} not found");
        }

        _context.Games.Remove(game);
        await _context.SaveChangesAsync();
    }

    public async Task<IEnumerable<Game>> GetGamesByGenreAsync(Guid genreId)
    {
        return await _context.Games
            .AsNoTracking()
            .Where(g => g.GenreId == genreId)
            .ToListAsync();
    }

    private async Task<bool> GameExistsAsync(Guid id)
    {
        return await _context.Games.AnyAsync(g => g.Id == id);
    }
}
```

### 8. Add Entity Framework Specific Endpoints

```csharp
// GET /games with include and filtering
app.MapGet("/games", async (
    IGameRepository repository,
    IGenreRepository genreRepository,
    string? genreName,
    decimal? minPrice,
    decimal? maxPrice,
    int? page,
    int? pageSize) =>
{
    // Get the base query from the repository
    var gamesQuery = await repository.GetAllAsync();

    // Apply filtering
    if (!string.IsNullOrWhiteSpace(genreName))
    {
        var genres = await genreRepository.GetAllAsync();
        var genre = genres.FirstOrDefault(g =>
            g.Name.Equals(genreName, StringComparison.OrdinalIgnoreCase));

        if (genre != null)
        {
            gamesQuery = gamesQuery.Where(g => g.GenreId == genre.Id);
        }
        else
        {
            return Results.Ok(Enumerable.Empty<Game>());
        }
    }

    if (minPrice.HasValue)
    {
        gamesQuery = gamesQuery.Where(g => g.Price >= minPrice.Value);
    }

    if (maxPrice.HasValue)
    {
        gamesQuery = gamesQuery.Where(g => g.Price <= maxPrice.Value);
    }

    // Apply pagination
    int effectivePageSize = pageSize ?? 10;
    int effectivePage = page ?? 1;

    var paginatedGames = gamesQuery
        .Skip((effectivePage - 1) * effectivePageSize)
        .Take(effectivePageSize);

    // Enrich with genre data
    var games = paginatedGames.ToList();
    var allGenres = await genreRepository.GetAllAsync();

    foreach (var game in games)
    {
        game.Genre = allGenres.FirstOrDefault(g => g.Id == game.GenreId);
    }

    return Results.Ok(games);
})
.WithName("GetAllGames")
.WithTags("Games");
```

### Database Migration Commands

```bash
# Create new migration
dotnet ef migrations add [MigrationName]

# Update database to latest migration
dotnet ef database update

# Revert to specific migration
dotnet ef database update [MigrationName]

# Remove last migration (if not applied)
dotnet ef migrations remove
```
