# GameStore API v2.0: Repository Pattern Implementation

## Overview

This document outlines the implementation plan for v2.0 of the GameStore API, which introduces the Repository pattern to abstract data access and prepare for future database integration.

## Key Concepts

The Repository pattern provides several benefits:

1. **Abstraction of data access logic** from business logic
2. **Unit testing** through dependency injection and mocking
3. **Consistent API** for accessing domain objects
4. **Preparation for multiple data sources** (in-memory, database, cloud)
5. **Separation of concerns** for better maintainability

### 2. Implement In-Memory Repositories

```csharp
namespace GameStore.Api.Repositories;

public class InMemoryGameRepository : IGameRepository
{
    private readonly List<Game> _games = new();

    public InMemoryGameRepository(IEnumerable<Game> initialGames)
    {
        _games.AddRange(initialGames);
    }

    public Task<IEnumerable<Game>> GetAllAsync()
    {
        return Task.FromResult(_games.AsEnumerable());
    }

    public Task<Game?> GetByIdAsync(Guid id)
    {
        return Task.FromResult(_games.FirstOrDefault(g => g.Id == id));
    }

    public Task<Game> CreateAsync(Game game)
    {
        game.Id = Guid.NewGuid();
        _games.Add(game);
        return Task.FromResult(game);
    }

    public Task UpdateAsync(Game game)
    {
        int index = _games.FindIndex(g => g.Id == game.Id);
        if (index == -1) throw new KeyNotFoundException($"Game with ID {game.Id} not found");

        _games[index] = game;
        return Task.CompletedTask;
    }

    public Task DeleteAsync(Guid id)
    {
        int index = _games.FindIndex(g => g.Id == id);
        if (index == -1) throw new KeyNotFoundException($"Game with ID {id} not found");

        _games.RemoveAt(index);
        return Task.CompletedTask;
    }

    public Task<IEnumerable<Game>> GetGamesByGenreAsync(Guid genreId)
    {
        return Task.FromResult(_games.Where(g => g.GenreId == genreId));
    }
}

public class InMemoryGenreRepository : IGenreRepository
{
    private readonly List<Genre> _genres = new();
    private readonly IGameRepository _gameRepository;

    public InMemoryGenreRepository(
        IEnumerable<Genre> initialGenres,
        IGameRepository gameRepository)
    {
        _genres.AddRange(initialGenres);
        _gameRepository = gameRepository;
    }

    public Task<IEnumerable<Genre>> GetAllAsync()
    {
        return Task.FromResult(_genres.AsEnumerable());
    }

    public Task<Genre?> GetByIdAsync(Guid id)
    {
        return Task.FromResult(_genres.FirstOrDefault(g => g.Id == id));
    }

    public Task<Genre> CreateAsync(Genre genre)
    {
        genre.Id = Guid.NewGuid();
        _genres.Add(genre);
        return Task.FromResult(genre);
    }

    public Task UpdateAsync(Genre genre)
    {
        int index = _genres.FindIndex(g => g.Id == genre.Id);
        if (index == -1) throw new KeyNotFoundException($"Genre with ID {genre.Id} not found");

        _genres[index] = genre;
        return Task.CompletedTask;
    }

    public Task DeleteAsync(Guid id)
    {
        int index = _genres.FindIndex(g => g.Id == id);
        if (index == -1) throw new KeyNotFoundException($"Genre with ID {id} not found");

        _genres.RemoveAt(index);
        return Task.CompletedTask;
    }

    public async Task<bool> GenreHasGamesAsync(Guid genreId)
    {
        var games = await _gameRepository.GetGamesByGenreAsync(genreId);
        return games.Any();
    }
}
```

### 5. Update Endpoints to Use Repositories

```csharp
// Games endpoints
app.MapGet("/games", async (IGameRepository repository, IGenreRepository genreRepository, string? genreName) =>
{
    IEnumerable<Game> games;

    if (!string.IsNullOrWhiteSpace(genreName))
    {
        // Find genre ID by name
        var genres = await genreRepository.GetAllAsync();
        var genre = genres.FirstOrDefault(g =>
            g.Name.Equals(genreName, StringComparison.OrdinalIgnoreCase));

        if (genre != null)
        {
            games = await repository.GetGamesByGenreAsync(genre.Id);
        }
        else
        {
            games = Enumerable.Empty<Game>();
        }
    }
    else
    {
        games = await repository.GetAllAsync();
    }

    // Enrich games with genre data
    var gamesList = new List<Game>();
    var genres = await genreRepository.GetAllAsync();

    foreach (var game in games)
    {
        game.Genre = genres.FirstOrDefault(g => g.Id == game.GenreId);
        gamesList.Add(game);
    }

    return gamesList;
})
.WithName("GetAllGames")
.WithTags("Games");

app.MapGet("/games/{id:guid}", async (IGameRepository repository, IGenreRepository genreRepository, Guid id) =>
{
    Game? game = await repository.GetByIdAsync(id);

    if (game is null)
    {
        return Results.NotFound();
    }

    // Get genre
    var genre = await genreRepository.GetByIdAsync(game.GenreId);
    game.Genre = genre;

    return Results.Ok(game);
})
.WithName("GetGameById")
.WithTags("Games");

app.MapPost("/games", async (IGameRepository repository, IGenreRepository genreRepository, Game game) =>
{
    // Verify genre exists
    var genre = await genreRepository.GetByIdAsync(game.GenreId);

    if (genre is null)
    {
        return Results.BadRequest($"Genre with ID {game.GenreId} does not exist");
    }

    var createdGame = await repository.CreateAsync(game);

    return Results.CreatedAtRoute(
        "GetGameById",
        new { id = createdGame.Id },
        createdGame);
})
.WithName("CreateGame")
.WithTags("Games")
.WithParameterValidation();
```

## Benefits of this Implementation

- **Testability**: Repositories can be mocked for unit testing
- **Separation of concerns**: Data access logic is separate from business logic
- **Future-proofing**: Easy transition to database storage in the future
- **Maintainability**: Consistent API for data access across the application
- **Code organization**: Better structure and organization of the project
